def run(label, argv):
    out, err = io.StringIO(), io.StringIO()
    try:
        with contextlib.redirect_stdout(out), contextlib.redirect_stderr(err):
            code = cli.main(argv)
        result = f"exit {code}"
    except BaseException as error:  # noqa: BLE001
        result = f"TRACEBACK {type(error).__name__}: {str(error)[:110]}"
    text = (err.getvalue() or out.getvalue()).strip().replace("\n", " / ")[:200]
    print(f"{label:45s} -> {result} {text}")
